/* linux/mm: simplified example */
#include <linux/mm.h>
#include <linux/errno.h>

#define MAX_BATCH 32

static int reclaim_page(struct page *page)
{
    if (PageDirty(page))
        return -EBUSY;

    clear_page_dirty_for_io(page);
    return 0;
}

static void reclaim_batch(struct page *pages[MAX_BATCH])
{
    int i;

    for (i = 0; i < MAX_BATCH; i++) {
        if (!pages[i])
            break;

        if (reclaim_page(pages[i]) == -EBUSY)
            pr_debug("mm: page %d busy\n", i);
    }
}
